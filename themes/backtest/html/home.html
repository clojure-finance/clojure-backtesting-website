(load-dataset "resources/sample-data/main" "main" add-aprc)
(init-portfolio "1990-01-02" 10000)

;; Hold 50 shares of AAA while its price
;; is above its 10-day moving average
(while (< (compare (get-date) "1990-03-29") 0)
  (let [price (get-permno-price "10001")
        ma10  (moving-avg "10001" 10)]
    (when (and price ma10)
      (order "10001" (if (> price ma10) 50 0)
             :remaining true)))
  (update-eval-report)
  (next-date))
(end-order)
